cmake_minimum_required(VERSION 3.5)
project(racecar_driver)

# Default to C++17
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

# Find dependencies
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)

SET(CMAKE_EXE_LINKER_FLAGS " -no-pie")

include_directories(
  include
)

add_executable(${PROJECT_NAME}_node src/racecar_driver_node.cpp src/art_racecar_driver.cpp)
add_executable(${PROJECT_NAME}_node_one src/racecar_driver_node_one.cpp src/art_racecar_driver.cpp)

ament_target_dependencies(${PROJECT_NAME}_node
  rclcpp
  geometry_msgs
  nav_msgs
  sensor_msgs
  std_msgs
  std_srvs
)

ament_target_dependencies(${PROJECT_NAME}_node_one
  rclcpp
  geometry_msgs
  nav_msgs
  sensor_msgs
  std_msgs
  std_srvs
)
